<script>
// ⚠️ ЗАМЕНИТЕ на публичный URL вашего backend после запуска туннеля
const API_URL = "https://ВАШ-ТУННЕЛЬ/ask";

const chat = document.getElementById("chat");
const form = document.getElementById("form");
const input = document.getElementById("q");
const send = document.getElementById("send");

function addMsg(role, html) {
  const m = document.createElement("div");
  m.className = "msg " + role;
  m.innerHTML = `<div class="avatar ${role}">${role === "bot" ? "CM" : "Я"}</div>` +
                `<div class="bubble">${html}</div>`;
  chat.appendChild(m);
  chat.scrollTop = chat.scrollHeight;
  return m;
}

function escapeHtml(s) {
  return s.replace(/[&<>"']/g, c => ({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#39;"}[c]));
}

form.addEventListener("submit", async (e) => {
  e.preventDefault();
  const text = input.value.trim();
  if (!text) return;
  input.value = "";
  input.disabled = send.disabled = true;

  addMsg("user", escapeHtml(text));
  const thinking = addMsg("bot", '<span class="typing">Думаю…</span>');

  try {
    const r = await fetch(API_URL, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ text })
    });
    if (!r.ok) throw new Error("HTTP " + r.status);
    const data = await r.json();

    let srcHtml = "";
    if (data.sources && data.sources.length) {
      const items = data.sources.map(s =>
        `<li><span class="file">[${s.n}] ${escapeHtml(s.file)}</span></li>`
      ).join("");
      srcHtml = `<details class="sources"><summary>Источники (${data.sources.length})</summary><ol>${items}</ol></details>`;
    }
    thinking.querySelector(".bubble").innerHTML =
      escapeHtml(data.answer).replace(/\n/g, "<br>") + srcHtml;
  } catch (err) {
    thinking.querySelector(".bubble").innerHTML =
      `<span style="color:#ff3b30">Ошибка: ${escapeHtml(err.message)}</span>`;
  } finally {
    input.disabled = send.disabled = false;
    input.focus();
  }
});

input.focus();
</script>
